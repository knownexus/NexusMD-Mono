# Profiles Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add fully-isolated switchable profiles to Helm — each with its own ADO connection (or no DevOps), planner data, team, and settings — plus a custom lists feature for manual (non-DevOps) profiles.

**Architecture:** localStorage keys are namespaced per-profile using a `getProfileKey(key)` helper (`helm.tweaks` → `helm.p.{id}.tweaks`). A one-time migration copies existing flat keys into the first profile on first launch. A React ProfileContext wraps the app and provides profile CRUD, export/import, and the active profile object. For manual profiles, ADO fetches are skipped and a Custom Lists view replaces the DevOps tabs.

**Tech Stack:** React 18 (global `React`/`ReactDOM`), Vite, Capacitor Android, no test framework — verify visually with `npm run dev`. PlModal/PlFld from `planner-ui.jsx` for sheets. Existing `localStorage` + JSON for storage.

**Reference spec:** `docs/superpowers/specs/2026-06-14-profiles-design.md`

```js
// profile-keys.js — localStorage key namespacing for profiles.
// No React, no Capacitor — safe to import from any file.

export function getProfileKey(key) {
  try {
    const id = localStorage.getItem('helm.activeProfileId');
    if (!id) return key;
    const suffix = key.startsWith('helm.') ? key.slice(5) : key;
    return `helm.p.${id}.${suffix}`;
  } catch (_) { return key; }
}

window.getProfileKey = getProfileKey;
```

## Task 2: profile-migration.js — first-run migration

```js
// profile-migration.js — one-time migration from flat helm.* keys to profile-scoped keys.
// Runs on every app boot; exits immediately after first successful migration.

function generateMigrationId() {
  return Math.random().toString(36).slice(2, 10) + Date.now().toString(36);
}

function runProfileMigration() {
  const ACTIVE_KEY  = 'helm.activeProfileId';
  const PROFILES_KEY = 'helm.profiles';
  const MIGRATED_KEY = 'helm.migrated';

  // Already migrated
  if (localStorage.getItem(ACTIVE_KEY)) return;

  const id = generateMigrationId();
  const prefix = `helm.p.${id}.`;

  // Determine profile name/type from existing ADO config
  const org = localStorage.getItem('helm.android.org') || '';

  // Collect all helm.* keys to copy (skip the 3 meta keys we're about to write)
  const skip = new Set([ACTIVE_KEY, PROFILES_KEY, MIGRATED_KEY]);
  const toCopy = [];
  for (let i = 0; i < localStorage.length; i++) {
    const k = localStorage.key(i);
    if (k && k.startsWith('helm.') && !skip.has(k)) toCopy.push(k);
  }

  // Copy to profile-scoped keys
  toCopy.forEach(k => {
    const suffix = k.slice('helm.'.length);
    localStorage.setItem(prefix + suffix, localStorage.getItem(k));
  });

  // Write profile metadata
  const profile = {
    id,
    name: org || 'My Workspace',
    initials: org ? org.slice(0, 2).toUpperCase() : 'MW',
    color: '#9a7e22',
    type: org ? 'ado' : 'manual',
    createdAt: Date.now(),
  };
  localStorage.setItem(PROFILES_KEY, JSON.stringify([profile]));
  localStorage.setItem(ACTIVE_KEY, id);
  localStorage.setItem(MIGRATED_KEY, String(Date.now()));
}

function pruneOldMigrationKeys() {
  const MIGRATED_KEY = 'helm.migrated';
  const migratedAt = parseInt(localStorage.getItem(MIGRATED_KEY) || '0', 10);
  if (!migratedAt) return;
  if (Date.now() - migratedAt < 30 * 24 * 60 * 60 * 1000) return;

  // 30 days have passed — safe to remove original unprefixed keys
  const keep = new Set(['helm.profiles', 'helm.activeProfileId', 'helm.migrated']);
  const toDelete = [];
  for (let i = 0; i < localStorage.length; i++) {
    const k = localStorage.key(i);
    if (k && k.startsWith('helm.') && !k.startsWith('helm.p.') && !keep.has(k)) {
      toDelete.push(k);
    }
  }
  toDelete.forEach(k => localStorage.removeItem(k));
  localStorage.removeItem(MIGRATED_KEY);
}

runProfileMigration();
pruneOldMigrationKeys();
```

```jsx
// profile-context.jsx — ProfileContext: active profile, CRUD, export/import.
// Reads/writes helm.profiles and helm.activeProfileId.
// getProfileKey (from profile-keys.js) handles per-profile storage namespacing.

const { useState: useSt, useCallback: useCb, useMemo: useMm, createContext, useContext, createElement: h } = React;

const PROFILES_KEY = 'helm.profiles';
const ACTIVE_KEY   = 'helm.activeProfileId';

function newId() {
  return Math.random().toString(36).slice(2, 10) + Date.now().toString(36);
}

function readProfiles() {
  try { return JSON.parse(localStorage.getItem(PROFILES_KEY) || '[]'); } catch (_) { return []; }
}

function writeProfiles(list) {
  localStorage.setItem(PROFILES_KEY, JSON.stringify(list));
}

const ProfileCtx = createContext(null);

function ProfileProvider({ children }) {
  const [profiles, setProfiles] = useSt(readProfiles);
  const [activeId, setActiveId] = useSt(() => localStorage.getItem(ACTIVE_KEY) || '');

  const activeProfile = profiles.find(p => p.id === activeId) || profiles[0] || null;

  const switchProfile = useCb((id) => {
    localStorage.setItem(ACTIVE_KEY, id);
    // Full reload so all useState initialisers re-run with the new profile's keys
    window.location.reload();
  }, []);

  const createProfile = useCb((opts) => {
    const p = {
      id: newId(),
      name:     opts.name     || 'New Profile',
      initials: opts.initials || (opts.name || 'NP').slice(0, 2).toUpperCase(),
      color:    opts.color    || '#9a7e22',
      type:     opts.type     || 'ado',
      createdAt: Date.now(),
    };
    const next = [...profiles, p];
    writeProfiles(next);
    setProfiles(next);
    return p;
  }, [profiles]);

  const updateProfile = useCb((id, changes) => {
    const next = profiles.map(p => p.id === id ? { ...p, ...changes } : p);
    writeProfiles(next);
    setProfiles(next);
  }, [profiles]);

  const deleteProfile = useCb((id) => {
    if (profiles.length <= 1) {
      window.toast?.('Cannot delete the only profile.', 'error');
      return;
    }
    // Remove all profile-scoped keys
    const pfx = `helm.p.${id}.`;
    const doomed = [];
    for (let i = 0; i < localStorage.length; i++) {
      const k = localStorage.key(i);
      if (k && k.startsWith(pfx)) doomed.push(k);
    }
    doomed.forEach(k => localStorage.removeItem(k));

    const next = profiles.filter(p => p.id !== id);
    writeProfiles(next);
    setProfiles(next);

    // If we deleted the active profile, switch to the first remaining one
    if (id === activeId && next.length > 0) {
      localStorage.setItem(ACTIVE_KEY, next[0].id);
      setActiveId(next[0].id);
      window.location.reload();
    }
  }, [profiles, activeId]);

  const exportProfile = useCb((id) => {
    const profile = profiles.find(p => p.id === id);
    if (!profile) return;
    const pfx = `helm.p.${id}.`;
    const data = {};
    for (let i = 0; i < localStorage.length; i++) {
      const k = localStorage.key(i);
      if (k && k.startsWith(pfx)) data[k.slice(pfx.length)] = localStorage.getItem(k);
    }
    const { name, initials, color, type } = profile;
    const blob = JSON.stringify({ version: 1, profile: { name, initials, color, type }, data }, null, 2);
    const url = URL.createObjectURL(new Blob([blob], { type: 'application/json' }));
    const a = document.createElement('a');
    a.href = url;
    a.download = `helm-profile-${name.replace(/[^a-z0-9]+/gi, '-').toLowerCase()}.json`;
    a.click();
    URL.revokeObjectURL(url);
  }, [profiles]);

  const importProfile = useCb((file) => {
    const reader = new FileReader();
    reader.onload = () => {
      try {
        const obj = JSON.parse(reader.result);
        if (obj.version !== 1 || !obj.profile || typeof obj.data !== 'object') {
          window.toast?.('Not a valid Helm profile file.', 'error');
          return;
        }
        const p = createProfile(obj.profile);
        const pfx = `helm.p.${p.id}.`;
        Object.entries(obj.data).forEach(([k, v]) => localStorage.setItem(pfx + k, v));
        window.toast?.(`Profile "${p.name}" imported.`, 'success');
      } catch (_) {
        window.toast?.('Not a valid Helm profile file.', 'error');
      }
    };
    reader.readAsText(file);
  }, [createProfile]);

  const value = useMm(() => ({
    profiles, activeProfile, activeId,
    switchProfile, createProfile, updateProfile, deleteProfile, exportProfile, importProfile,
  }), [profiles, activeProfile, activeId, switchProfile, createProfile, updateProfile, deleteProfile, exportProfile, importProfile]);

  return h(ProfileCtx.Provider, { value }, children);
}

function useProfile() {
  return useContext(ProfileCtx);
}

window.ProfileProvider = ProfileProvider;
window.useProfile = useProfile;
```

```jsx
    if (def.id === 'status' && v) {
      const bucket = toBucket(v);
      const bg = bucket === 'done' ? '#e8f5e9' : bucket === 'blocked' ? '#fce4ec' : bucket === 'inprogress' ? '#fff3e0' : 'var(--surface-2)';
      const col = bucket === 'done' ? '#2e7d32' : bucket === 'blocked' ? '#c62828' : bucket === 'inprogress' ? '#e65100' : 'var(--text)';
      return <span style={{ fontSize: 11, padding: '2px 8px', borderRadius: 99, background: bg, color: col }}>{v}</span>;
    }
    return v || '—';
  };

  return (
    <div>
      <div style={{ display: 'flex', alignItems: 'center', gap: 8, padding: '10px 0', marginBottom: 4 }}>
        <div style={{ flex: 1 }} />
        <button className="btn sm" onClick={() => setEditItem('new')}>+ Add item</button>
      </div>

      {items.length === 0 && (
        <div style={{ padding: '40px 0', textAlign: 'center', opacity: .4, fontSize: 14 }}>
          No items yet — click "+ Add item" to get started.
        </div>
      )}

      {items.map(item => (
        <div key={item.id} style={{ borderBottom: '1px solid var(--border)' }}>
          <div style={{ display: 'flex', alignItems: 'center', gap: 8, padding: '10px 0' }}>
            <button className="btn icon ghost sm" style={{ opacity: .4 }} onClick={() => toggleExpand(item.id)}>
              {expandedIds.has(item.id) ? '▼' : '▶'}
            </button>
            <div style={{ flex: 1, fontWeight: 600, fontSize: 13 }}>{item.title}</div>
            {fieldDefs.map(def => (
              <div key={def.id} style={{ fontSize: 12, minWidth: 60, textAlign: 'right' }}>
                {renderCell(item, def)}
              </div>
            ))}
            <button className="btn icon ghost sm" onClick={() => setEditItem(item)} style={{ opacity: .5 }}>
              <Icon name="edit" size={12} />
            </button>
            <button className="btn icon ghost sm danger" onClick={() => deleteItem(item.id)} style={{ opacity: .4 }}>
              <Icon name="x" size={12} />
            </button>
          </div>

          {expandedIds.has(item.id) && (
            <div style={{ marginLeft: 28, marginBottom: 10, display: 'flex', flexDirection: 'column', gap: 4 }}>
              {(item.children || []).map(sub => (
                <SubItemRow key={sub.id} sub={sub}
                  onToggle={(sid) => toggleSub(item.id, sid)}
                  onDelete={(sid) => deleteSub(item.id, sid)} />
              ))}
              <div style={{ display: 'flex', alignItems: 'center', gap: 6, marginTop: 4 }}>
                <input
                  className="mock-input" placeholder="Add sub-item…"
                  value={newSubText[item.id] || ''}
                  style={{ flex: 1, fontSize: 12 }}
                  onChange={e => setNewSubText(prev => ({ ...prev, [item.id]: e.target.value }))}
                  onKeyDown={e => { if (e.key === 'Enter') addSubItem(item.id, newSubText[item.id] || ''); }}
                />
                <button className="btn sm" onClick={() => addSubItem(item.id, newSubText[item.id] || '')}>Add</button>
              </div>
            </div>
          )}
        </div>
      ))}

      {editingItem && (
        <ItemEditor
          item={editingItem === 'new' ? null : editingItem}
          list={list}
          onSave={saveItem}
          onClose={() => setEditItem(null)} />
      )}
    </div>
  );
}
```

```jsx
function ListsView() {
  const [data, setData] = useLSt(readLists);
  const [activeListId, setActiveListId] = useLSt(null);
  const [mainTab, setMainTab] = useLSt('list');  // 'list' | 'board' | 'mine'
  const [editingList, setEditList] = useLSt(null); // list object or 'new'

  const save = (next) => { setData(next); writeLists(next); };

  const createList = (opts) => {
    const l = { id: newId(), items: [], ...opts };
    const next = { lists: [...data.lists, l] };
    save(next);
    setActiveListId(l.id);
    setEditList(null);
  };

  const updateList = (id, changes) => {
    save({ lists: data.lists.map(l => l.id === id ? { ...l, ...changes } : l) });
  };

  const deleteList = (id) => {
    if (!window.confirm('Delete this list and all its items?')) return;
    const next = { lists: data.lists.filter(l => l.id !== id) };
    save(next);
    if (activeListId === id) setActiveListId(next.lists[0]?.id || null);
  };

  const activeList = data.lists.find(l => l.id === activeListId) || data.lists[0] || null;

  return (
    <React.Fragment>
      <div className="m-appbar">
        <div className="m-ab-lead">
          <h1>Lists</h1>
          <div className="m-ab-sub">Custom tracking for your team</div>
        </div>
        <div className="m-ab-actions">
          <button className="btn sm" onClick={() => setEditList('new')}>+ New list</button>
        </div>
      </div>
      <div className="m-screen" data-screen-label="Lists">
        <div style={{ display: 'flex', gap: 0, minHeight: 0, flex: 1 }}>

          {/* List sidebar */}
          <div style={{ width: 180, flexShrink: 0, borderRight: '1px solid var(--border)', padding: '12px 8px', display: 'flex', flexDirection: 'column', gap: 2 }}>
            {data.lists.map(l => (
              <div key={l.id} style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
                <button
                  className={`pl-rail-item${activeList?.id === l.id ? ' active' : ''}`}
                  style={{ flex: 1, textAlign: 'left' }}
                  onClick={() => { setActiveListId(l.id); setMainTab('list'); }}>
                  <span style={{ color: l.color, marginRight: 4 }}>●</span>{l.name}
                </button>
                <button className="btn icon ghost sm" style={{ opacity: .4, flexShrink: 0 }}
                  onClick={() => setEditList(l)}><Icon name="edit" size={11}/></button>
              </div>
            ))}
            {data.lists.length === 0 && (
              <div style={{ fontSize: 12, opacity: .4, padding: '8px 4px' }}>No lists yet.</div>
            )}
          </div>

          {/* Main content */}
          <div style={{ flex: 1, padding: '16px 20px', minWidth: 0, overflowY: 'auto' }}>

            {/* Top tab bar */}
            <div style={{ display: 'flex', gap: 0, marginBottom: 16, borderBottom: '1px solid var(--border)' }}>
              {[['list', activeList ? activeList.name : 'Items'], ['board', 'Board'], ['mine', 'My items']].map(([id, label]) => (
                <button key={id} onClick={() => setMainTab(id)} style={{
                  padding: '6px 16px', fontSize: 13, border: 'none', background: 'none',
                  borderBottom: mainTab === id ? '2px solid var(--accent)' : '2px solid transparent',
                  color: mainTab === id ? 'var(--accent)' : 'var(--text)', fontWeight: mainTab === id ? 600 : 400,
                  cursor: 'pointer',
                }}>{label}</button>
              ))}
            </div>

            {mainTab === 'list' && activeList && (
              <ListDetailView
                list={activeList}
                onUpdate={(changes) => updateList(activeList.id, changes)} />
            )}
            {mainTab === 'list' && !activeList && (
              <div style={{ padding: '60px 0', textAlign: 'center', opacity: .4 }}>
                Create a list to get started.
              </div>
            )}
            {mainTab === 'board' && <BoardView lists={data.lists} />}
            {mainTab === 'mine'  && <MyItemsView lists={data.lists} />}
          </div>
        </div>
      </div>

      {/* List editor modal */}
      {editingList && (
        <ListEditor
          list={editingList === 'new' ? null : editingList}
          onSave={(opts) => {
            if (editingList === 'new') {
              createList(opts);
            } else {
              updateList(editingList.id, opts);
              setEditList(null);
            }
          }}
          onClose={() => setEditList(null)} />
      )}
    </React.Fragment>
  );
}
```
